# CredPilot — runtime dependencies.
#
# Everything installs with pip. No Docker, no external database service, no
# hosted model or vector endpoint (REQ-032).
#
#   python -m venv .venv
#   .venv/Scripts/activate      # Windows
#   source .venv/bin/activate   # macOS / Linux
#   pip install -r requirements.txt
#
# Versions are the ones the committed evidence in eval/results/ and
# data/vectorstore/ was produced with.

# --- language and agent framework (REQ-035) ---------------------------------
# Python 3.11+ is required; see pyproject.toml.
langgraph==1.2.11
langgraph-checkpoint-sqlite==3.1.1
langchain-core==1.6.2

# --- memory (REQ-038) -------------------------------------------------------
# The source names the checkpointer and LangMem together. The checkpointer above
# carries a single thread; LangMem carries what survives between threads, over a
# LangGraph SQLite store. Used in src/memory/semantic.py, and every long-term
# write goes through it.
langmem==0.0.30

# --- LLM provider: Google Gemini only (REQ-036) -----------------------------
# Not used in the retrieval, calculation or rule-evaluation path, which are
# deterministic by design. Wired for narrative generation and LLM-as-judge.
langchain-google-genai==4.4.0

# --- interoperability (REQ-037) ---------------------------------------------
mcp==1.30.0
langchain-mcp-adapters>=0.3.0,<0.4

# --- web application ----------------------------------------------------------
# The same graph over HTTP. Both arrive transitively today (FastAPI via Chroma,
# uvicorn via Phoenix), and both are pinned here anyway: a surface the project
# ships should not depend on another package continuing to want it.
fastapi==0.141.1
uvicorn==0.51.0
# Pinned because leaving it to FastAPI's `starlette>=0.46.0` resolved to 1.3.1,
# which Arize Phoenix 11.38.0 predates: it calls Starlette's
# `Jinja2Templates.TemplateResponse()` with the pre-1.0 signature, so every
# Phoenix HTML route answered 500 and its UI could not be opened at all. The
# API still served spans, which is what made it easy to miss. Pinning here
# rather than upgrading Phoenix nine major versions at the end of the project;
# FastAPI 0.141.1 accepts this range and `tests/test_web_api.py` passes on it.
starlette==0.52.1

# --- retrieval: Chroma + local Sentence-Transformers (REQ-039) --------------
chromadb==1.5.9
sentence-transformers==6.0.1
torch==2.14.0
transformers==5.17.0
# Lexical layer. Pure Python, no service (architecture decision D-04).
rank-bm25==0.2.2

# --- observability: Arize Phoenix + OpenTelemetry / OpenInference (REQ-040) --
arize-phoenix==11.38.0

opentelemetry-sdk==1.44.0

# --- security and PII (REQ-042) ---------------------------------------------
# The source names "Guardrails-AI / LLM Guard - Presidio (PII) - python-dotenv".
#
# Guardrails-AI rather than LLM Guard, and the reason is a hard constraint rather
# than a preference: llm-guard pins transformers==4.51.3, while
# sentence-transformers 6.0.1 requires transformers>=5.0.0. Installing it would
# downgrade the embedding stack the whole retrieval path and the memory index
# run on. Guardrails-AI resolves against the existing pins.
#
# Its schema validators run locally and call no model. Its anonymous telemetry is
# switched off in code (src/guardrails/policy_guard.py) rather than by a machine
# local ~/.guardrailsrc, because a PII system must not ship spans to a vendor
# endpoint on someone else's laptop either.
guardrails-ai==0.11.0
presidio-analyzer==2.2.361
presidio-anonymizer==2.2.361
python-dotenv==1.2.2
